### 3.2 Pinia 스토어

#### stores/todo.ts

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { todoApi } from '@/api/todo'
import type { Todo, TodoFilter, GroupedTodos, TodoCreateRequest, TodoUpdateRequest } from '@/types/todo'

export const useTodoStore = defineStore('todo', () => {
  // State
  const todos = ref<Todo[]>([])
  const sharedTodos = ref<Todo[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)
  const currentFilter = ref<TodoFilter>('all')
  const searchKeyword = ref('')

  // Getters
  const groupedTodos = computed<GroupedTodos>(() => {
    const today = new Date().toISOString().split('T')[0]
    const tomorrow = new Date(Date.now() + 86400000).toISOString().split('T')[0]

    const myTodos = todos.value.filter(t => t.isOwner)
    const shared = todos.value.filter(t => !t.isOwner)
    const transferred = myTodos.filter(t => t.transferredFromUserName)
    const owned = myTodos.filter(t => !t.transferredFromUserName)

    return {
      today: owned.filter(t => t.dueDate === today && !t.isCompleted),
      tomorrow: owned.filter(t => t.dueDate === tomorrow && !t.isCompleted),
      upcoming: owned.filter(t => t.dueDate && t.dueDate > tomorrow && !t.isCompleted),
      overdue: owned.filter(t => t.dueDate && t.dueDate < today && !t.isCompleted),
      noDate: owned.filter(t => !t.dueDate && !t.isCompleted),
      completed: owned.filter(t => t.isCompleted),
      shared: shared,
      transferred: transferred
    }
  })

  const todayCount = computed(() => groupedTodos.value.today.length)
  const overdueCount = computed(() => groupedTodos.value.overdue.length)
  const totalActiveCount = computed(() =>
    todos.value.filter(t => !t.isCompleted).length
  )

  // Actions
  async function fetchTodos(filter?: TodoFilter) {
    loading.value = true
    error.value = null
    try {
      const params = {
        filter: filter || currentFilter.value,
        includeShared: true,
        includeCompleted: filter === 'completed',
        keyword: searchKeyword.value || undefined
      }
      const response = await todoApi.getTodos(params)
      if (response.success && response.data) {
        todos.value = response.data.content
      }
    } catch (e) {
      error.value = '할 일 목록을 불러오는데 실패했습니다.'
    } finally {
      loading.value = false
    }
  }

  async function createTodo(request: TodoCreateRequest) {
    const response = await todoApi.createTodo(request)
    if (response.success && response.data) {
      todos.value.unshift(response.data)
      return response.data
    }
    throw new Error(response.message)
  }

  async function updateTodo(todoId: number, request: TodoUpdateRequest) {
    const response = await todoApi.updateTodo(todoId, request)
    if (response.success && response.data) {
      const index = todos.value.findIndex(t => t.todoId === todoId)
      if (index !== -1) {
        todos.value[index] = { ...todos.value[index], ...response.data }
      }
      return response.data
    }
    throw new Error(response.message)
  }

  async function deleteTodo(todoId: number) {
    const response = await todoApi.deleteTodo(todoId)
    if (response.success) {
      todos.value = todos.value.filter(t => t.todoId !== todoId)
    }
  }

  async function toggleComplete(todoId: number, isCompleted: boolean) {
    const response = await todoApi.toggleComplete(todoId, isCompleted)
    if (response.success && response.data) {
      const index = todos.value.findIndex(t => t.todoId === todoId)
      if (index !== -1) {
        todos.value[index] = { ...todos.value[index], ...response.data }
      }
    }
  }

  async function shareTodo(todoId: number, sharedUserId: number, permission: string) {
    return await todoApi.addShare(todoId, { sharedUserId, permission })
  }

  async function transferTodo(todoId: number, toUserId: number) {
    const response = await todoApi.transferTodo(todoId, toUserId)
    if (response.success) {
      // 이관 후 목록에서 제거
      todos.value = todos.value.filter(t => t.todoId !== todoId)
    }
  }

  function setFilter(filter: TodoFilter) {
    currentFilter.value = filter
    fetchTodos(filter)
  }

  function setSearchKeyword(keyword: string) {
    searchKeyword.value = keyword
    fetchTodos()
  }

  return {
    // State
    todos,
    sharedTodos,
    loading,
    error,
    currentFilter,
    searchKeyword,

    // Getters
    groupedTodos,
    todayCount,
    overdueCount,
    totalActiveCount,

    // Actions
    fetchTodos,
    createTodo,
    updateTodo,
    deleteTodo,
    toggleComplete,
    shareTodo,
    transferTodo,
    setFilter,
    setSearchKeyword
  }
})
```

#### ChecklistSection.vue (업무 상세용)

```vue
<script setup lang="ts">
/**
 * 업무 내 체크리스트 섹션
 */
import { ref, onMounted, computed } from 'vue'
import { checklistApi } from '@/api/checklist'
import ChecklistItem from './ChecklistItem.vue'
import ChecklistForm from './ChecklistForm.vue'
import type { Checklist } from '@/types/checklist'

const props = defineProps<{
  itemId: number
}>()

const checklist = ref<Checklist[]>([])
const totalCount = ref(0)
const completedCount = ref(0)
const loading = ref(false)
const showAddForm = ref(false)

const progressPercent = computed(() => {
  if (totalCount.value === 0) return 0
  return Math.round((completedCount.value / totalCount.value) * 100)
})

onMounted(() => {
  fetchChecklist()
})

async function fetchChecklist() {
  loading.value = true
  try {
    const response = await checklistApi.getChecklist(props.itemId)
    if (response.success && response.data) {
      checklist.value = response.data.items
      totalCount.value = response.data.totalCount
      completedCount.value = response.data.completedCount
    }
  } finally {
    loading.value = false
  }
}

async function handleToggleComplete(id: number, isCompleted: boolean) {
  await checklistApi.toggleComplete(id, isCompleted)
  fetchChecklist()
}

async function handleDelete(id: number) {
  await checklistApi.deleteChecklist(id)
  fetchChecklist()
}

async function handleCreated() {
  showAddForm.value = false
  fetchChecklist()
}
</script>

<template>
  <div class="border-t border-gray-200 dark:border-gray-700 pt-4 mt-4">
    <!-- 헤더 -->
    <div class="flex items-center justify-between mb-3">
      <div class="flex items-center gap-2">
        <span class="text-sm font-medium text-gray-700 dark:text-gray-300">
          ☑️ 체크리스트
        </span>
        <span class="text-xs text-gray-500">
          ({{ completedCount }}/{{ totalCount }})
        </span>
      </div>
      <button
        type="button"
        class="text-xs text-primary-600 hover:text-primary-700"
        @click="showAddForm = !showAddForm"
      >
        + 항목 추가
      </button>
    </div>

    <!-- 진행률 바 -->
    <div v-if="totalCount > 0" class="h-1.5 bg-gray-200 dark:bg-gray-700 rounded-full mb-3">
      <div
        class="h-full bg-green-500 rounded-full transition-all"
        :style="{ width: `${progressPercent}%` }"
      />
    </div>

    <!-- 추가 폼 -->
    <ChecklistForm
      v-if="showAddForm"
      :item-id="itemId"
      @created="handleCreated"
      @cancel="showAddForm = false"
    />

    <!-- 체크리스트 목록 -->
    <div class="space-y-1">
      <ChecklistItem
        v-for="item in checklist"
        :key="item.checklistId"
        :item="item"
        @toggle-complete="handleToggleComplete(item.checklistId, $event)"
        @delete="handleDelete(item.checklistId)"
      />
    </div>

    <!-- 빈 상태 -->
    <div v-if="!loading && checklist.length === 0 && !showAddForm" class="text-center py-4">
      <p class="text-sm text-gray-400">체크리스트가 없습니다.</p>
    </div>
  </div>
</template>
```
